-- d

local f = workspace.guessTheFlag.FlagPart.flagDecal
local fol = workspace.gtfFlags

-- Configuration
local ALLOWED_USERS = {"coolcat8520", "kamlol18"}
local currentUser = game.Players.LocalPlayer.Name

-- Check if user is allowed
if not table.find(ALLOWED_USERS, currentUser) then
    error("User not authorized to run this script")
end

local otherUser = (currentUser == "coolcat8520") and "kamlol18" or "coolcat8520"
local canGuess = false
local otherJustMessaged = false
local loopActive = false

function find(t)
    for _,v in pairs(fol:GetChildren()) do
        if v:IsA("Decal") and v.Texture == t then
            return v.Name
        end
    end
    return nil
end

function send(txt)
    local rs = game:GetService("ReplicatedStorage")
    local c = rs:FindFirstChild("DefaultChatSystemChatEvents")
    
    local function retry()
        if c then
            local s = c:FindFirstChild("SayMessageRequest")
            if s then
                local ok = pcall(function()
                    s:FireServer(txt, "All")
                end)
                if not ok then
                    wait(2)
                    retry()
                end
            end
        else
            local t = game:GetService("TextChatService")
            local ch = t.TextChannels:FindFirstChild("RBXGeneral")
            if ch then
                local ok = pcall(function()
                    ch:SendAsync(txt)
                end)
                if not ok then
                    wait(2)
                    retry()
                end
            end
        end
    end
    
    retry()
end

-- Listen for chat messages
local function setupChatListener()
    local rs = game:GetService("ReplicatedStorage")
    local c = rs:FindFirstChild("DefaultChatSystemChatEvents")
    
    if c then
        local me = c:FindFirstChild("OnMessageDoneFiltering")
        if me then
            me.OnClientEvent:Connect(function(data)
                local speaker = data.FromSpeaker
                local message = data.Message
                
                -- Check for !start command from coolcat8520
                if speaker == "coolcat8520" and message == "!start" and not loopActive then
                    loopActive = true
                    print("Starting loop...")
                    
                    -- kamlol18 sends first message
                    if currentUser == "kamlol18" then
                        wait(0.5)
                        local tex = f.Texture
                        local name = find(tex)
                        if name then
                            send(string.lower(name))
                        end
                        wait(0.1)
                        -- Start the guessing loop for both users
                        coroutine.wrap(startGuessing)()
                    elseif currentUser == "coolcat8520" then
                        -- coolcat8520 just starts listening
                        coroutine.wrap(startGuessing)()
                    end
                end
                
                -- Track when other user sends any message (for the loop)
                if speaker == otherUser and loopActive then
                    otherJustMessaged = true
                end
            end)
        end
    else
        local t = game:GetService("TextChatService")
        local ch = t.TextChannels:FindFirstChild("RBXGeneral")
        if ch then
            ch.MessageReceived:Connect(function(message)
                if message.TextSource and message.TextSource.Name == "coolcat8520" and message.Text == "!start" and not loopActive then
                    loopActive = true
                    print("Starting loop...")
                    
                    if currentUser == "kamlol18" then
                        wait(0.5)
                        local tex = f.Texture
                        local name = find(tex)
                        if name then
                            send(string.lower(name))
                        end
                        wait(0.1)
                        coroutine.wrap(startGuessing)()
                    elseif currentUser == "coolcat8520" then
                        coroutine.wrap(startGuessing)()
                    end
                end
                
                if message.TextSource and message.TextSource.Name == otherUser and loopActive then
                    otherJustMessaged = true
                end
            end)
        end
    end
end

-- Main guessing loop
function startGuessing()
    while loopActive do
        -- Wait for the other person to message
        otherJustMessaged = false
        repeat
            wait(0.1)
        until otherJustMessaged == true or not loopActive
        
        if not loopActive then break end
        
        -- Wait 0.75 seconds after their message
        wait(0.75)
        
        -- Send the current flag
        local tex = f.Texture
        local name = find(tex)
        if name then
            send(string.lower(name))
        end
        
        -- Brief pause before listening again
        wait(0.1)
    end
end

-- Setup and wait for !start command
setupChatListener()

print("Script loaded. Waiting for coolcat8520 to type '!start' in chat...")

-- Keep script running
while not loopActive do
    wait(1)
end
