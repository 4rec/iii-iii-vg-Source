-- c

local f = workspace.guessTheFlag.FlagPart.flagDecal
local fol = workspace.gtfFlags

-- Configuration
local ALLOWED_USERS = {"coolcat8520", "kamlol18"}
local currentUser = game.Players.LocalPlayer.Name

-- Check if user is allowed
if not table.find(ALLOWED_USERS, currentUser) then
    error("User not authorized to run this script")
end

local otherUser = (currentUser == "coolcat8520") and "kamlol18" or "coolcat8520"
local isActive = false
local otherLastTyped = tick()
local otherListenerActive = false

-- Debouncing variables
local lastSentTexture = nil
local lastSendTime = 0
local pendingSend = nil
local sendDebounceTime = 0.5 -- Don't send same texture within 0.5 seconds

function find(t)
    for _,v in pairs(fol:GetChildren()) do
        if v:IsA("Decal") and v.Texture == t then
            return v.Name
        end
    end
    return nil
end

function send(txt)
    local rs = game:GetService("ReplicatedStorage")
    local c = rs:FindFirstChild("DefaultChatSystemChatEvents")
    
    local function retry()
        if c then
            local s = c:FindFirstChild("SayMessageRequest")
            if s then
                local ok = pcall(function()
                    s:FireServer(txt, "All")
                end)
                if not ok then
                    wait(2)
                    retry()
                end
            end
        else
            local t = game:GetService("TextChatService")
            local ch = t.TextChannels:FindFirstChild("RBXGeneral")
            if ch then
                local ok = pcall(function()
                    ch:SendAsync(txt)
                end)
                if not ok then
                    wait(2)
                    retry()
                end
            end
        end
    end
    
    retry()
end

function sendWithDelay(txt)
    -- Cancel pending send if it's the same texture
    if pendingSend then
        pendingSend:Disconnect()
        pendingSend = nil
    end
    
    -- Schedule the send after 1.5 seconds
    local connection
    connection = game:GetService("RunService").Stepped:Connect(function()
        connection:Disconnect()
        pendingSend = nil
        send(txt)
    end)
    pendingSend = connection
end

function startGuessing()
    if isActive then return end
    isActive = true
    
    -- Reset debounce when starting
    lastSentTexture = nil
    lastSendTime = 0
    
    -- Get initial flag
    local initialName = find(f.Texture)
    if initialName then
        sendWithDelay(string.lower(initialName))
        lastSentTexture = f.Texture
        lastSendTime = tick()
    end
    
    -- Main guessing loop
    f:GetPropertyChangedSignal("Texture"):Connect(function()
        if isActive then
            local tex = f.Texture
            local currentTime = tick()
            
            -- Check if this texture was recently sent
            if lastSentTexture == tex and (currentTime - lastSendTime) < sendDebounceTime then
                return -- Skip duplicate
            end
            
            local name = find(tex)
            if name then
                lastSentTexture = tex
                lastSendTime = currentTime
                sendWithDelay(string.lower(name))
            end
        end
    end)
end

function stopGuessing()
    isActive = false
    -- Clear pending send
    if pendingSend then
        pendingSend:Disconnect()
        pendingSend = nil
    end
end

-- Check for cooldown on the other person
function monitorOtherUser()
    while true do
        wait(1) -- Check every second
        local timeSinceOtherTyped = tick() - otherLastTyped
        
        -- If other user hasn't typed in 5 seconds and we're not currently active
        if timeSinceOtherTyped >= 5 and not isActive and not otherListenerActive then
            otherListenerActive = true
            print("Other user cooldown detected, activating guesser...")
            startGuessing()
            
            -- Give the other user time to come back, then deactivate
            wait(8) -- Let guesser run for 8 seconds
            stopGuessing()
            otherListenerActive = false
        end
    end
end

-- Listen for other user's chat messages
local function setupChatListener()
    local rs = game:GetService("ReplicatedStorage")
    local c = rs:FindFirstChild("DefaultChatSystemChatEvents")
    
    if c then
        local me = c:FindFirstChild("OnMessageDoneFiltering")
        if me then
            me.OnClientEvent:Connect(function(data)
                if data.FromSpeaker == otherUser then
                    otherLastTyped = tick()
                    -- If we're currently guessing, stop and let other user take over
                    if isActive then
                        stopGuessing()
                        otherListenerActive = false
                        print("Other user started typing, stopping guesser...")
                    end
                end
            end)
        end
    else
        local t = game:GetService("TextChatService")
        local ch = t.TextChannels:FindFirstChild("RBXGeneral")
        if ch then
            ch.MessageReceived:Connect(function(message)
                if message.TextSource and message.TextSource.Name == otherUser then
                    otherLastTyped = tick()
                    if isActive then
                        stopGuessing()
                        otherListenerActive = false
                        print("Other user started typing, stopping guesser...")
                    end
                end
            end)
        end
    end
end

-- Initial typing for kamlol18
if currentUser == "kamlol18" then
    wait(2) -- Wait a moment for game to load
    send("a")
    wait(1) -- Let the message send
end

-- Start monitoring the other user
coroutine.wrap(monitorOtherUser)()
setupChatListener()

-- If you're coolcat8520, start guessing immediately (kamlol18 will start the cycle)
if currentUser == "coolcat8520" then
    wait(1)
    otherLastTyped = tick() + 6 -- Pretend kamlol hasn't typed for 6 seconds
    wait(5) -- Wait for kamlol's initial message to pass
end
